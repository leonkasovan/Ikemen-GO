# Makefile for Ikemen-GO (Windows/MinGW targets)
# Requires MSYS2 MINGW64 environment with:
#   mingw-w64-x86_64-toolchain mingw-w64-x86_64-go
#   mingw-w64-x86_64-SDL2 mingw-w64-x86_64-libxmp
#   mingw-w64-x86_64-nasm mingw-w64-x86_64-yasm
#
# Usage:
#   make              # Build Win64 release (default)
#   make debug=1      # Build Win64 debug (console subsystem)
#   make cpp=1        # Build Win64 with C++ modules (IKEMEN_CPP)
#   make Win32        # Build Win32
#   make ffmpeg=yes   # Force local FFmpeg build
#   make clean        # Remove build artifacts
#   make distclean    # Remove build artifacts + FFmpeg build dir + screenpack

SHELL := /usr/bin/bash
.SHELLFLAGS := -euo pipefail -c
.ONESHELL:

# ---- Project metadata ----
APP_VERSION   ?= nightly
APP_BUILDTIME ?= $(shell date '+%Y.%m.%d')
COPY_START_YEAR := 2016

# Compute SXS_VERSION at make-parse time (avoids quoting nightmares in recipes).
# Strip leading v/V, keep only digits+dots, pad/clamp to 4 numeric parts.
_sxs_clean   = $(shell v='$(strip $(subst v,,$(subst V,,$(1))))'; \
  [[ "$$v" =~ ^[0-9.]+$$ ]] || { echo "0.0.0.0"; exit; }; \
  IFS='.' read -ra p <<<"$$v"; \
  for ((i=$${#p[@]}; i<4; i++)); do p+=("0"); done; \
  out=(); for x in "$${p[@]:0:4}"; do \
    [[ "$$x" =~ ^[0-9]+$$ ]] || x=0; ((x<0)) && x=0; ((x>65535)) && x=65535; \
    out+=("$$x"); \
  done; IFS='.'; echo "$${out[*]}")
SXS_VERSION := $(call _sxs_clean,$(APP_VERSION))

# Decompose SXS_VERSION into components for VERSIONINFO
_sxs_major := $(word 1,$(subst ., ,$(SXS_VERSION)))
_sxs_minor := $(word 2,$(subst ., ,$(SXS_VERSION)))
_sxs_patch := $(word 3,$(subst ., ,$(SXS_VERSION)))
_sxs_build := $(word 4,$(subst ., ,$(SXS_VERSION)))

# ---- Targets ----
TARGET        ?= Win64
OUTDIR        ?= .
LIBDIR        ?= lib
BUILDDIR      ?= build
DELAYLIB_DIR   = $(BUILDDIR)/delaylib
FFMPEG_PREFIX ?= $(CURDIR)/$(BUILDDIR)/ffmpeg
FFMPEG_REV    ?= release/7.1
FFMPEG_SRCDIR := $(BUILDDIR)/ffmpeg-src

# ---- Build flags ----
debug         ?= 0
cpp           ?= 0
ffmpeg        ?= auto   # auto|yes|no

# ---- Set defaults based on TARGET ----
ifeq ($(TARGET),Win64)
  GOOS    := windows
  GOARCH  := amd64
  CROSS_PREFIX := x86_64-w64-mingw32
  BINNAME := $(if $(filter 1,$(cpp)),Ikemen_CPP.exe,Ikemen_GO.exe)
else ifeq ($(TARGET),Win32)
  GOOS    := windows
  GOARCH  := 386
  CROSS_PREFIX := i686-w64-mingw32
  BINNAME := Ikemen_GO_x86.exe
endif

# Compiler (allow override from CLI or environment)
CC  ?= $(CROSS_PREFIX)-gcc
CXX ?= $(CROSS_PREFIX)-g++

# Go build tags
GO_TAGS := $(and $(filter 1,$(cpp)),ikemen_cpp)

# Linker flags
GO_LDFLAGS := -s -w
GO_LDFLAGS += -X 'main.Version=$(APP_VERSION)' -X 'main.BuildTime=$(APP_BUILDTIME)'
ifeq ($(debug),1)
  BUILD_TYPE := debug
else
  GO_LDFLAGS := -H windowsgui $(GO_LDFLAGS)
  BUILD_TYPE := release
endif

# windres detection: try prefixed name first, then bare (matches build.sh logic)
WINDRES := $(or \
  $(shell command -v $(CROSS_PREFIX)-windres 2>/dev/null), \
  $(shell command -v windres 2>/dev/null))

# windres target flag
WINDRES_TARGET := $(if $(filter 386,$(GOARCH)),--target=pe-i386,--target=pe-x86-64)

# ---- pkg-config and dependency detection ----
PKG_CONFIG ?= pkg-config

# Prepend local FFmpeg prefix to PKG_CONFIG_PATH if directory exists
FFMPEG_PC_DIR := $(FFMPEG_PREFIX)/lib/pkgconfig
ifeq ($(shell test -d '$(FFMPEG_PC_DIR)' && echo yes),yes)
  export PKG_CONFIG_PATH := $(FFMPEG_PC_DIR):$(PKG_CONFIG_PATH)
endif

# Detect if FFmpeg is available via pkg-config (used by _ffmpeg target)
_HAS_FFMPEG := $(shell $(PKG_CONFIG) --exists libavformat libavcodec libavutil libswresample libswscale libavfilter 2>/dev/null && echo yes)

# Derived CGO flags — evaluated after PKG_CONFIG_PATH is set
CGO_CFLAGS  := $(shell $(PKG_CONFIG) --cflags libavformat libavcodec libavutil libswscale libswresample libavfilter libxmp sdl2 2>/dev/null || echo "")
CGO_LIBS    := $(shell $(PKG_CONFIG) --libs   libavformat libavcodec libavutil libswscale libswresample libavfilter libxmp sdl2 2>/dev/null || echo "")
CGO_LDFLAGS := -L$(CURDIR)/$(DELAYLIB_DIR) $(CGO_LIBS)

export CGO_ENABLED  := 1
export GOEXPERIMENT := arenas
export GOOS
export GOARCH
export CC
export CXX
export CGO_CFLAGS
export CGO_LDFLAGS

# ---- Phony targets ----
.PHONY: all Win64 Win32 debug release clean distclean ffmpeg screenpack bundle FORCE

all: Win64

# ===========================================================================
# Windows resource generation
# ===========================================================================

# Generate .rc and .manifest with fresh version data every build (FORCE)
build/winres/Ikemen_GO.rc: FORCE
	@echo "==> Generating Windows version resources..."
	mkdir -p build/winres
	SXS="$(SXS_VERSION)"
	VMAJ="$(_sxs_major)"
	VMIN="$(_sxs_minor)"
	VPAT="$(_sxs_patch)"
	VREV="$(_sxs_build)"
	YEAR="$(APP_BUILDTIME)"; YEAR="$${YEAR%%-*}"
	COPY="(c) $(COPY_START_YEAR)-$${YEAR} Ikemen GO team (MIT)"
	ARCH="$(if $(filter amd64,$(GOARCH)),amd64,x86)"
	cat > build/winres/Ikemen_GO.exe.manifest <<-MANEOF
	<?xml version="1.0" encoding="UTF-8" standalone="yes"?>
	<assembly xmlns="urn:schemas-microsoft-com:asm.v1" manifestVersion="1.0">
	  <assemblyIdentity type="win32" name="Ikemen_GO" version="$${SXS}" processorArchitecture="$${ARCH}"/>
	  <dependency>
	    <dependentAssembly>
	      <assemblyIdentity type="win32" name="Microsoft.Windows.Common-Controls"
	        version="6.0.0.0" processorArchitecture="*" publicKeyToken="6595b64144ccf1df" language="*"/>
	    </dependentAssembly>
	  </dependency>
	</assembly>
	MANEOF
	cat > build/winres/Ikemen_GO.rc <<-RCEOF
	#include <windows.h>
	#include <winver.h>
	1 ICON "Ikemen_Cylia_V2.ico"
	1 RT_MANIFEST "Ikemen_GO.exe.manifest"

	VS_VERSION_INFO VERSIONINFO
	 FILEVERSION $(_sxs_major),$(_sxs_minor),$(_sxs_patch),$(_sxs_build)
	 PRODUCTVERSION $(_sxs_major),$(_sxs_minor),$(_sxs_patch),$(_sxs_build)
	 FILEFLAGSMASK 0x3fL
	 FILEFLAGS 0x0L
	 FILEOS 0x4L
	 FILETYPE 0x1L
	 FILESUBTYPE 0x0L
	BEGIN
	    BLOCK "StringFileInfo"
	    BEGIN
	        BLOCK "040904B0"
	        BEGIN
	            VALUE "CompanyName", "Ikemen GO\\0"
	            VALUE "FileDescription", "Ikemen GO\\0"
	            VALUE "FileVersion", "$${SXS}\\0"
	            VALUE "ProductName", "Ikemen GO\\0"
	            VALUE "ProductVersion", "$${SXS}\\0"
	            VALUE "OriginalFilename", "Ikemen_GO.exe\\0"
	            VALUE "InternalName", "Ikemen_GO\\0"
	            VALUE "BuildDate", "$(APP_BUILDTIME)\\0"
	            VALUE "LegalCopyright", "$${COPY}\\0"
	        END
	    END
	    BLOCK "VarFileInfo"
	    BEGIN
	        VALUE "Translation", 0x0409, 1200
	    END
	END
	RCEOF

# Compile .rc -> .syso (Go auto-links .syso files in the package directory)
src/rsrc_windows.syso: build/winres/Ikemen_GO.rc
	@echo "==> Embedding Windows resources (icon + manifest)..."
	mkdir -p src
	"$(WINDRES)" $(WINDRES_TARGET) \
	  -I build/winres -I external/icons \
	  -i build/winres/Ikemen_GO.rc \
	  -O coff -o $@

# ===========================================================================
# Delay-load import libs for MinGW (Windows)
# ===========================================================================

$(DELAYLIB_DIR)/.stamp:
	@echo "==> Generating delay-load import libs..."
	mkdir -p $(DELAYLIB_DIR)
	shopt -s nullglob
	for d in "$(FFMPEG_PREFIX)"/bin/*.dll /mingw64/bin/libxmp*.dll /mingw64/bin/libwinpthread-1.dll /mingw64/bin/SDL2*.dll; do
		[[ -f "$$d" ]] || continue
		base="$$(basename "$$d")"
		name="$${base%.dll}"
		libname="$${name%%-*}"
		libname="$${libname#lib}"
		(cd "$(DELAYLIB_DIR)" && gendef "$$d" >/dev/null)
		dlltool --dllname "$$base" --def "$(DELAYLIB_DIR)/$${name}.def" --output-delaylib "$(DELAYLIB_DIR)/lib$${libname}.dll.a"
		rm -f "$(DELAYLIB_DIR)/$${name}.def"
	done
	shopt -u nullglob
	touch $@

# ===========================================================================
# FFmpeg local build (conditional)
# ===========================================================================

.PHONY: ffmpeg
ffmpeg:
	$(MAKE) _ffmpeg

.PHONY: _ffmpeg
_ffmpeg:
	@echo "==> Checking FFmpeg..."
ifeq ($(ffmpeg),no)
	@echo "==> FFmpeg build skipped (ffmpeg=no)."
else ifeq ($(_HAS_FFMPEG),yes)
	@echo "==> FFmpeg found via pkg-config; using it."
else
	@echo "==> Building minimal FFmpeg ($(FFMPEG_REV))..."
	rm -rf $(FFMPEG_SRCDIR)
	git clone --depth=1 -b "$(FFMPEG_REV)" https://github.com/FFmpeg/FFmpeg.git $(FFMPEG_SRCDIR)
	cd $(FFMPEG_SRCDIR) \
	  && ./configure \
	    --prefix="$(FFMPEG_PREFIX)" \
	    --install-name-dir=@rpath \
	    --enable-shared --disable-static \
	    --disable-gpl --disable-nonfree \
	    --disable-debug --disable-doc --disable-programs --disable-everything \
	    --disable-autodetect \
	    --enable-avformat --enable-avcodec --enable-avutil --enable-swresample --enable-swscale \
	    --enable-avfilter --enable-filter=buffer,buffersink,format,scale,pad,crop \
	    --enable-protocol=file \
	    --enable-demuxer=matroska,webm \
	    --enable-decoder=vp8,vp9,opus,vorbis \
	    --enable-parser=vp8,vp9,opus,vorbis \
	    --cc="$(CC)" \
	    --pkg-config="$(shell which pkg-config)" \
	  && make -j"$$(getconf _NPROCESSORS_ONLN || echo 2)" \
	  && make install
	@echo "==> FFmpeg built and installed to: $(FFMPEG_PREFIX)"
endif

# ===========================================================================
# Screenpack clone/update
# ===========================================================================

SCREENPACK_REPO ?= https://github.com/ikemen-engine/Ikemen-GO-Screenpack.git
SCREENPACK_REF  ?= master
SCREENPACK_DIR   = $(BUILDDIR)/screenpack

.PHONY: screenpack
screenpack:
	@echo "==> Ensuring Elecbyte screenpack..."
	mkdir -p $(BUILDDIR)
	if [ ! -d "$(SCREENPACK_DIR)/.git" ]; then
		rm -rf $(SCREENPACK_DIR)
		git clone --depth=1 -b "$(SCREENPACK_REF)" "$(SCREENPACK_REPO)" "$(SCREENPACK_DIR)"
	else
		cd "$(SCREENPACK_DIR)" \
		  && git fetch --depth=1 origin "$(SCREENPACK_REF)" \
		  && git checkout -f FETCH_HEAD
	fi

# ===========================================================================
# Bundle shared DLLs next to the binary
# ===========================================================================

.PHONY: bundle
bundle:
	@echo "==> Bundling shared libraries..."
	mkdir -p $(LIBDIR)
	if [ -d "$(FFMPEG_PREFIX)/bin" ]; then
		cp -av "$(FFMPEG_PREFIX)"/bin/*.dll "$(LIBDIR)/" 2>/dev/null || true
	fi
	for d in \
		/mingw64/bin/libwinpthread-1.dll \
		/mingw64/bin/libgcc_s_seh-1.dll \
		/mingw64/bin/libstdc++-6.dll \
		/mingw64/bin/libxmp*.dll \
		/mingw64/bin/SDL2*.dll; do
		cp -av "$$d" "$(LIBDIR)/" 2>/dev/null || true
	done
	@echo "==> Libraries bundled in $(LIBDIR)/"

# ===========================================================================
# Main build
# ===========================================================================

$(BUILDDIR):
	mkdir -p $(BUILDDIR)

$(BINNAME): $(BUILDDIR) build/winres/Ikemen_GO.rc | _ffmpeg
	$(MAKE) src/rsrc_windows.syso
	$(MAKE) $(DELAYLIB_DIR)/.stamp
	@echo "==> Building $(BINNAME) ($(BUILD_TYPE), GOARCH=$(GOARCH))..."
	$(if $(GO_TAGS),GOFLAGS="-tags=$(GO_TAGS)" ;) \
	go build -trimpath -v \
	  -ldflags "$(GO_LDFLAGS)" \
	  -o "$(OUTDIR)/$(BINNAME)" ./src
	rm -f src/rsrc_windows.syso
	$(MAKE) bundle
	@echo "==> Build successful: $(OUTDIR)/$(BINNAME)"
	@test -d "$(LIBDIR)" && echo "    Runtime DLLs: $(LIBDIR)/" || true

# ===========================================================================
# Convenience targets
# ===========================================================================

Win64:
	+$(MAKE) TARGET=Win64 $(BINNAME)

Win32:
	+$(MAKE) TARGET=Win32 $(BINNAME)

debug:
	+$(MAKE) debug=1 TARGET=$(TARGET) $(BINNAME)

release:
	+$(MAKE) debug=0 TARGET=$(TARGET) $(BINNAME)

# ===========================================================================
# Clean
# ===========================================================================

clean:
	@echo "==> Cleaning build artifacts..."
	rm -f "$(OUTDIR)/Ikemen_GO.exe" "$(OUTDIR)/Ikemen_CPP.exe" "$(OUTDIR)/Ikemen_GO_x86.exe"
	rm -f src/rsrc_windows.syso
	rm -rf $(DELAYLIB_DIR)
	rm -rf $(LIBDIR)
	rm -rf build/winres
	@echo "==> Clean done."

distclean: clean
	@echo "==> Deep cleaning..."
	rm -rf $(FFMPEG_SRCDIR)
	rm -rf $(FFMPEG_PREFIX)
	rm -rf $(SCREENPACK_DIR)
	@echo "==> Distclean done."
