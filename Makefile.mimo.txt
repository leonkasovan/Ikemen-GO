# Ikemen-GO Windows Makefile (MSYS2 / MinGW64)
# Derived from build/build.sh — covers the Win64 build path.
#
# Usage:
#   make                  # Release build (default)
#   make debug            # Debug build (console subsystem)
#   make IKEMEN_CPP=1     # C++ conversion build
#   make clean            # Remove build artifacts
#   make distclean        # Remove build artifacts + ffmpeg build dir
#
# Prerequisites (MSYS2 MINGW64 shell):
#   pacman -S --noconfirm git make diffutils mingw-w64-x86_64-pkg-config \
#     mingw-w64-x86_64-go mingw-w64-x86_64-toolchain \
#     mingw-w64-x86_64-nasm mingw-w64-x86_64-yasm \
#     mingw-w64-x86_64-tools-git mingw-w64-x86_64-libxmp \
#     mingw-w64-x86_64-SDL2

# ─── Paths ───────────────────────────────────────────────────────────────────
BUILDDIR  := build
LIBDIR    := lib
OUTDIR    := .
DELAYLIB_DIR := $(BUILDDIR)/delaylib

FFMPEG_SRCDIR := $(BUILDDIR)/ffmpeg-src
FFMPEG_REV    ?= release/7.1
FFMPEG_PREFIX ?= $(abspath $(BUILDDIR)/ffmpeg)

# ─── Options ─────────────────────────────────────────────────────────────────
DEBUG_BUILD   ?= 0
IKEMEN_CPP    ?= 0
BUILD_FFMPEG  ?= auto        # auto | yes | no
APP_VERSION   ?= nightly
APP_BUILDTIME ?= $(shell date '+%Y.%m.%d')

# ─── Toolchain ───────────────────────────────────────────────────────────────
export GOEXPERIMENT := arenas
export CGO_ENABLED  := 1
export CGO_LDFLAGS  ?=

# MSYS2 / MinGW64 paths
export PATH    := /mingw64/bin:$(PATH)
export GOROOT ?= /mingw64/lib/go

# Use plain tool names in native MINGW64 (cross-names only for cross-compilation)
CC  ?= gcc
CXX ?= g++
WINDRES ?= $(shell command -v x86_64-w64-mingw32-windres 2>/dev/null || command -v windres 2>/dev/null || echo windres)

# PKG_CONFIG
PKG_CONFIG ?= pkg-config

# Packages for pkg-config
PKG_PKGS := libavformat libavcodec libavutil libswscale libswresample libavfilter libxmp sdl2

# Output binary name
ifeq ($(IKEMEN_CPP),1)
  BIN_NAME := Ikemen_CPP.exe
else
  BIN_NAME := Ikemen_GO.exe
endif

# Version ldflags
LDFLAGS_BASE := -s -w \
  -X 'main.Version=$(APP_VERSION)' \
  -X 'main.BuildTime=$(APP_BUILDTIME)'

# Debug vs Release
ifeq ($(DEBUG_BUILD),1)
  LDFLAGS_GO := $(LDFLAGS_BASE)
else
  LDFLAGS_GO := -H windowsgui $(LDFLAGS_BASE)
endif

# Build tags
TAGS :=
ifeq ($(IKEMEN_CPP),1)
  TAGS := -tags ikemen_cpp
endif

# ─── Phony targets ───────────────────────────────────────────────────────────
.PHONY: all debug release clean distclean \
        check-deps deps delaylibs resources binary bundle help

# ─── Default target ──────────────────────────────────────────────────────────
all: check-deps deps delaylibs resources binary bundle
	@echo "==> Build successful"
	@echo "    Binary: $(OUTDIR)/$(BIN_NAME)"
	@if [ -d "$(LIBDIR)" ]; then echo "    Runtime DLLs: $(LIBDIR)/"; fi

release: all

debug:
	$(MAKE) all DEBUG_BUILD=1

# ─── Dependency checks ──────────────────────────────────────────────────────
check-deps:
	@echo "==> Checking build dependencies..."
	@command -v git      >/dev/null 2>&1 || { echo "ERROR: git not found" >&2; exit 1; }
	@command -v make     >/dev/null 2>&1 || { echo "ERROR: make not found" >&2; exit 1; }
	@command -v $(PKG_CONFIG) >/dev/null 2>&1 || { echo "ERROR: pkg-config not found" >&2; exit 1; }
	@command -v gcc      >/dev/null 2>&1 || { echo "ERROR: gcc not found (install mingw-w64-x86_64-toolchain)" >&2; exit 1; }
	@command -v g++      >/dev/null 2>&1 || { echo "ERROR: g++ not found" >&2; exit 1; }
	@command -v nasm     >/dev/null 2>&1 || { echo "ERROR: nasm not found" >&2; exit 1; }
	@command -v go       >/dev/null 2>&1 || { echo "ERROR: go not found (install mingw-w64-x86_64-go)" >&2; exit 1; }
	@command -v gendef   >/dev/null 2>&1 || { echo "ERROR: gendef not found (install mingw-w64-x86_64-tools-git)" >&2; exit 1; }
	@command -v dlltool  >/dev/null 2>&1 || { echo "ERROR: dlltool not found" >&2; exit 1; }
	@echo "    All dependencies found."

# ─── FFmpeg + pkg-config setup ───────────────────────────────────────────────
deps: check-deps
ifeq ($(BUILD_FFMPEG),yes)
	@echo "==> BUILD_FFMPEG=yes: forcing local FFmpeg build"
	@$(MAKE) _build-ffmpeg
	@export PKG_CONFIG_PATH="$(FFMPEG_PREFIX)/lib/pkgconfig:$${PKG_CONFIG_PATH:-}"; \
	$(MAKE) _check-sdl2 _check-libxmp
else ifeq ($(BUILD_FFMPEG),no)
	@echo "==> BUILD_FFMPEG=no: using system FFmpeg"
	@$(PKG_CONFIG) --exists libavformat libavcodec libavutil libswresample libswscale libavfilter 2>/dev/null || \
		{ echo "ERROR: FFmpeg not found via pkg-config." >&2; exit 1; }
	@$(MAKE) _check-sdl2 _check-libxmp
else
	@# auto: try system first, build local if missing
	@if $(PKG_CONFIG) --exists libavformat libavcodec libavutil libswresample libswscale libavfilter 2>/dev/null; then \
		echo "==> Found FFmpeg via pkg-config; using it."; \
		$(MAKE) _check-sdl2 _check-libxmp; \
	else \
		echo "==> FFmpeg not found; building minimal local copy."; \
		$(MAKE) _build-ffmpeg; \
		export PKG_CONFIG_PATH="$(FFMPEG_PREFIX)/lib/pkgconfig:$${PKG_CONFIG_PATH:-}"; \
		$(MAKE) _check-sdl2 _check-libxmp; \
	fi
endif

_check-sdl2:
	@echo "==> Checking SDL2..."
	@$(PKG_CONFIG) --exists sdl2 2>/dev/null || \
		{ echo "ERROR: SDL2 not found. Install: pacman -S mingw-w64-x86_64-SDL2" >&2; exit 1; }
	@echo "    SDL2 found."

_check-libxmp:
	@echo "==> Checking libxmp..."
	@$(PKG_CONFIG) --exists libxmp 2>/dev/null || \
		{ echo "ERROR: libxmp not found. Install: pacman -S mingw-w64-x86_64-libxmp" >&2; exit 1; }
	@echo "    libxmp found."

# Internal: build FFmpeg from source
_build-ffmpeg: $(FFMPEG_SRCDIR)/.git
	@echo "==> Building FFmpeg to $(FFMPEG_PREFIX)..."
	cd $(FFMPEG_SRCDIR) && \
		./configure \
			--prefix="$(FFMPEG_PREFIX)" \
			--install-name-dir=@rpath \
			--enable-shared --disable-static \
			--disable-gpl --disable-nonfree \
			--disable-debug --disable-doc --disable-programs --disable-everything \
			--disable-autodetect \
			--enable-avformat --enable-avcodec --enable-avutil --enable-swresample --enable-swscale \
			--enable-avfilter --enable-filter=buffer,buffersink,format,scale,pad,crop \
			--enable-protocol=file \
			--enable-demuxer=matroska,webm \
			--enable-decoder=vp8,vp9,opus,vorbis \
			--enable-parser=vp8,vp9,opus,vorbis && \
	$(MAKE) -j$$(nproc 2>/dev/null || echo 2) && \
	$(MAKE) install

$(FFMPEG_SRCDIR)/.git:
	@echo "==> Cloning FFmpeg $(FFMPEG_REV)..."
	@mkdir -p $(BUILDDIR)
	git clone --depth=1 -b $(FFMPEG_REV) https://github.com/FFmpeg/FFmpeg.git $(FFMPEG_SRCDIR)

# ─── Delay-load import libraries ────────────────────────────────────────────
delaylibs:
	@echo "==> Creating delay-load import libraries..."
	@mkdir -p $(DELAYLIB_DIR)
	@found=0; \
	for dll in \
		$(FFMPEG_PREFIX)/bin/*.dll \
		/mingw64/bin/libxmp*.dll \
		/mingw64/bin/libwinpthread-1.dll \
		/mingw64/bin/SDL2*.dll; do \
		[ -f "$$dll" ] || continue; \
		found=1; \
		base=$$(basename "$$dll"); \
		name=$${base%.dll}; \
		libname=$${name%%-*}; \
		libname=$${libname#lib}; \
		(cd "$(DELAYLIB_DIR)" && gendef "$$dll" >/dev/null 2>&1); \
		dlltool --dllname "$$base" \
			--def "$(DELAYLIB_DIR)/$$name.def" \
			--output-delaylib "$(DELAYLIB_DIR)/lib$$libname.dll.a"; \
		rm -f "$(DELAYLIB_DIR)/$$name.def"; \
	done; \
	if [ "$$found" = "0" ]; then \
		echo "    No DLLs found for delay-load import libs (expected if no local FFmpeg)."; \
	fi

# ─── Windows resources (icon + manifest + version) ──────────────────────────
# Sanitize version to "A.B.C.D" (all numeric). Invalids → 0.0.0.0
SXS_VERSION := $(shell echo "$(APP_VERSION)" | sed 's/^[vV]//' | sed 's/[^0-9].*//' | awk -F. '{ printf "%d.%d.%d.%d", $$1+0, $$2+0, $$3+0, $$4+0 }')
VMAJ := $(word 1,$(subst ., ,$(SXS_VERSION)))
VMIN := $(word 2,$(subst ., ,$(SXS_VERSION)))
VPAT := $(word 3,$(subst ., ,$(SXS_VERSION)))
VREV := $(word 4,$(subst ., ,$(SXS_VERSION)))

ASM_ARCH  := amd64
WINDRES_TARGET := -F pe-x86-64

COPY_START_YEAR ?= 2016
BUILD_YEAR       := $(subst -,,$(firstword $(subst ., ,$(APP_BUILDTIME))))
APP_COPYRIGHT    ?= (c) $(COPY_START_YEAR)-$(BUILD_YEAR) Ikemen GO team (MIT)

resources: check-deps
	@echo "==> Embedding Windows resources (icon + manifest)..."
	@mkdir -p $(BUILDDIR)/winres src
	@# Generate manifest
	@printf '%s\n' \
		'<?xml version="1.0" encoding="UTF-8" standalone="yes"?>' \
		'<assembly xmlns="urn:schemas-microsoft-com:asm.v1" manifestVersion="1.0">' \
		'  <assemblyIdentity type="win32" name="Ikemen_GO" version="$(SXS_VERSION)" processorArchitecture="$(ASM_ARCH)"/>' \
		'  <dependency>' \
		'    <dependentAssembly>' \
		'      <assemblyIdentity type="win32" name="Microsoft.Windows.Common-Controls"' \
		'        version="6.0.0.0" processorArchitecture="*" publicKeyToken="6595b64144ccf1df" language="*"/>' \
		'    </dependentAssembly>' \
		'  </dependency>' \
		'</assembly>' \
		> "$(BUILDDIR)/winres/Ikemen_GO.exe.manifest"
	@# Generate RC
	@printf '%s\n' \
		'#include <windows.h>' \
		'#include <winver.h>' \
		'1 ICON "Ikemen_Cylia_V2.ico"' \
		'1 RT_MANIFEST "Ikemen_GO.exe.manifest"' \
		'' \
		'VS_VERSION_INFO VERSIONINFO' \
		' FILEVERSION $(VMAJ),$(VMIN),$(VPAT),$(VREV)' \
		' PRODUCTVERSION $(VMAJ),$(VMIN),$(VPAT),$(VREV)' \
		' FILEFLAGSMASK 0x3fL' \
		' FILEFLAGS 0x0L' \
		' FILEOS 0x4L' \
		' FILETYPE 0x1L' \
		' FILESUBTYPE 0x0L' \
		'BEGIN' \
		'    BLOCK "StringFileInfo"' \
		'    BEGIN' \
		'        BLOCK "040904B0"' \
		'        BEGIN' \
		'            VALUE "CompanyName", "Ikemen GO\\0"' \
		'            VALUE "FileDescription", "Ikemen GO\\0"' \
		'            VALUE "FileVersion", "$(SXS_VERSION)\\0"' \
		'            VALUE "ProductName", "Ikemen GO\\0"' \
		'            VALUE "ProductVersion", "$(SXS_VERSION)\\0"' \
		'            VALUE "OriginalFilename", "Ikemen_GO.exe\\0"' \
		'            VALUE "InternalName", "Ikemen_GO\\0"' \
		'            VALUE "BuildDate", "$(APP_BUILDTIME)\\0"' \
		'            VALUE "LegalCopyright", "$(APP_COPYRIGHT)\\0"' \
		'        END' \
		'    END' \
		'    BLOCK "VarFileInfo"' \
		'    BEGIN' \
		'        VALUE "Translation", 0x0409, 1200' \
		'    END' \
		'END' \
		> "$(BUILDDIR)/winres/Ikemen_GO.rc"
	@# Compile RC -> COFF
	$(WINDRES) $(WINDRES_TARGET) \
		-I "$(BUILDDIR)/winres" -I external/icons \
		-i "$(BUILDDIR)/winres/Ikemen_GO.rc" \
		-O coff -o src/rsrc_windows.syso
	@echo "    Windows resources embedded."

# ─── Go binary ──────────────────────────────────────────────────────────────
binary: check-go-env resources
	@echo "==> Building Go binary (this may take a while)..."
	@# Compute CGO flags from pkg-config (with local FFmpeg PKG_CONFIG_PATH if needed)
	@_pc_path=""; \
	if [ -d "$(FFMPEG_PREFIX)/lib/pkgconfig" ]; then \
		_pc_path="$(FFMPEG_PREFIX)/lib/pkgconfig:"; \
	fi && \
	export PKG_CONFIG_PATH="$${_pc_path}$${PKG_CONFIG_PATH:-}" && \
	export CGO_CFLAGS="$$($(PKG_CONFIG) --cflags $(PKG_PKGS))" && \
	export CGO_LDFLAGS="-L$$PWD/$(DELAYLIB_DIR) $$($(PKG_CONFIG) --libs $(PKG_PKGS)) $${CGO_LDFLAGS:-}" && \
	go build -trimpath -v $(TAGS) \
		-ldflags "$(LDFLAGS_GO)" \
		-o "$(OUTDIR)/$(BIN_NAME)" ./src
	@# Clean embedded resource object
	@rm -f src/rsrc_windows.syso

# ─── Bundle shared libs ─────────────────────────────────────────────────────
bundle:
	@echo "==> Bundling runtime DLLs to $(LIBDIR)/..."
	@mkdir -p $(LIBDIR)
	@# FFmpeg DLLs
	@if [ -d "$(FFMPEG_PREFIX)/bin" ]; then \
		cp -av $(FFMPEG_PREFIX)/bin/*.dll $(LIBDIR)/ 2>/dev/null || true; \
	fi
	@# MSYS2 system DLLs
	@for dll in \
		/mingw64/bin/libwinpthread-1.dll \
		/mingw64/bin/libgcc_s_seh-1.dll \
		/mingw64/bin/libstdc++-6.dll \
		/mingw64/bin/libxmp*.dll \
		/mingw64/bin/SDL2*.dll; do \
		[ -f "$$dll" ] && cp -av "$$dll" $(LIBDIR)/ 2>/dev/null || true; \
	done

# ─── Clean ──────────────────────────────────────────────────────────────────
clean:
	@echo "==> Cleaning build artifacts..."
	rm -f src/rsrc_windows.syso
	rm -f $(OUTDIR)/$(BIN_NAME)
	rm -f $(OUTDIR)/Ikemen_GO.exe
	rm -f $(OUTDIR)/Ikemen_CPP.exe
	rm -rf $(LIBDIR)
	rm -rf $(DELAYLIB_DIR)
	rm -rf $(BUILDDIR)/winres

distclean: clean
	@echo "==> Removing FFmpeg build directory..."
	rm -rf $(FFMPEG_SRCDIR)
	rm -rf $(BUILDDIR)/ffmpeg

# ─── Go environment validation ──────────────────────────────────────────────
check-go-env:
	@go version >/dev/null 2>&1 || \
		{ echo "ERROR: 'go version' failed. Set GOROOT=/mingw64/lib/go" >&2; exit 1; }
